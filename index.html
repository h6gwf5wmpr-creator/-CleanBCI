function runMindAnalysis() { let rawInput = document.getElementById('eeg-input').value.trim(); // تنظيف أي رموز غير مقبولة وتعديل الأقواس والفاصلة العربية تلقائياً لضمان دقة النظام البرمجي let sanitized = rawInput .replace(/،/g, ',') .replace(/[“”]/g, '"') .trim(); // التأكد من وجود أقواس المصفوفة، وإضافتها تلقائياً إن لم تكن موجودة لتبقى الأمور احترافية if (!sanitized.startsWith('[')) sanitized = '[' + sanitized; if (!sanitized.endsWith(']')) sanitized = sanitized + ']'; let features; try { features = JSON.parse(sanitized); } catch (e) { alert('الرجاء التأكد من كتابة المصفوفة والأقواس بشكل صحيح!'); return; } if (!Array.isArray(features) || features.length === 0) { alert('الرجاء التأكد من كتابة المصفوفة والأقواس بشكل صحيح!'); return; } const res = processBrainMind(features); document.getElementById('neuro-classification').innerText = res.classification; document.getElementById('avg-signal').innerText = res.avgSignal; document.getElementById('power-signal').innerText = res.powerSignal; document.getElementById('analysis-ar').innerText = res.descAr; document.getElementById('analysis-en').innerText = res.descEn; document.getElementById('brain-wave-visualizer').style.opacity = "1"; document.querySelectorAll('[id^="card-"]').forEach(el => { el.classList.remove('opacity-100', 'border-indigo-500', 'bg-indigo-950/40', 'scale-105', 'shadow-lg'); el.classList.add('opacity-40'); }); if (res.cardId) { const activeCard = document.getElementById(res.cardId); if (activeCard) { activeCard.classList.remove('opacity-40'); activeCard.classList.add('opacity-100', 'border-indigo-500', 'bg-indigo-950/40', 'scale-105', 'shadow-lg'); } } const now = new Date(); const timeString = now.toLocaleTimeString('en-US', { hour: '2-digit', minute: '2-digit', second: '2-digit' }); diagnosticHistory.unshift({ time: timeString, classification: res.classification }); if (diagnosticHistory.length > 3) diagnosticHistory.pop(); let logHtml = ""; diagnosticHistory.forEach(item => { logHtml += `
${item.classification} ${item.time}
`; }); document.getElementById('history-log').innerHTML = logHtml; }
