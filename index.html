function processBrainMind(signalArray) { if (!Array.isArray(signalArray) || signalArray.length === 0) { return { classification: "Idle State", descAr: "لا توجد إشارة", descEn: "No signal", avgSignal: "0.00 µV", powerSignal: "0.00 dB" }; } const maxVal = Math.max(...signalArray); const minVal = Math.min(...signalArray); const sum = signalArray.reduce((a, b) => a + b, 0); const avg = sum / signalArray.length; const len = signalArray.length; // حساب المدى المطلق للإشارة لمعرفة شدة التذبذب const signalRange = maxVal - minVal; let result = { classification: "", descAr: "", descEn: "", avgSignal: `${avg.toFixed(2)} µV`, powerSignal: `${(Math.abs(sum) * 1.5).toFixed(2)} dB` }; // خريطة الفصل بين الإشارات البشرية الحقيقية if (maxVal > 30.0) { result.classification = "Right Hand Movement / حركة اليد اليمنى"; result.descAr = "تم رصد تفريغ عصبي حاد وموجه في القشرة الحركية اليسرى يتجاوز 30 ميكروفولت، مما يشير إلى أمر حركي مؤكد ومباشر لتحريك اليد اليمنى."; result.descEn = "A sharp, localized neural discharge exceeding 30 microvolts detected in the left motor cortex, indicating a definitive motor command to move the Right Hand."; } else if (maxVal > 15.0 && maxVal <= 30.0 && signalRange < 50.0) { result.classification = "Left Hand Movement / حركة اليد اليسرى"; result.descAr = "تم رصد تفجر إشاري نشط في القشرة الحركية اليمنى بمدى مستقر، مما يؤكد نجاح فك تشفير أمر تحريك اليد اليسرى وتجاوز عائق الجمجمة."; result.descEn = "An active signal burst registered within the right motor cortex with stable range, validating successful decoding of the Left Hand movement command."; } else if (signalRange > 70.0) { result.classification = "Interactive Gaming & Reflexes / طور اللعب والتفاعل السريع"; result.descAr = "تذبذب حاد جداً وعنيف في الإشارات الكهربائية بفارق يتجاوز 70 ميكروفولت بين القمة والقاع، يحاكي استجابات ردود الفعل الفورية أثناء ألعاب الفيديو."; result.descEn = "High-amplitude visual-motor oscillations exceeding 70 microvolts range, perfectly simulating instant reflex actions and rapid decision-making during gaming."; } else if (avg > 8.0 && len >= 8) { result.classification = "Spontaneous Walking / المشي وحركة الأطراف السفلية"; result.descAr = "إشارات متزامنة واسعة النطاق وممتدة عبر عدة قنوات بجهد مستمر، تتوافق مع الأنماط الحركية التلقائية والعفوية لتنسيق حركة الأرجل والمشي."; result.descEn = "Broad, multi-channel synchronized signals with continuous voltage, corresponding to automated motor patterns for leg coordination and walking."; } else if (maxVal > 0 && maxVal <= 6.0 && minVal >= -5.0) { result.classification = "Cognitive Focus & Reading / التركيز الذهني والقراءة"; result.descAr = "إشارة منخفضة الجهد ومستقرة جداً (موجات بيتا الهادئة بحدود 4 ميكروفولت) فوق مناطق المعالجة البصرية، تدل على القراءة والتركيز الذهني العميق."; result.descEn = "Low-voltage, highly stable signal (Beta waves around 4 microvolts) over visual processing centers, demonstrating active text comprehension and deep focus."; } else if (avg < 0 && minVal < -1.0) { result.classification = "Biological Excretion Urge / الرغبة في قضاء الحاجة"; result.descAr = "فك تشفير نبضات بطيئة ذات جهد سالب قادمة من قاع الدماغ (الجهاز العصبي الذاتي)، تحاكي بدقة التنبيهات الحشوية الداخلية والرغبة في قضاء الحاجة."; result.descEn = "Successful decoding of low-frequency, negative-voltage autonomic visceral impulses, tracking internal physiological indicators and the urge to evacuate."; } else { result.classification = "Spontaneous Reflex / حركة ارتدادية عفوية عامة"; result.descAr = "نشاط قشري عام وخلفي يتوافق مع الحركات الدقيقة واللاإرادية التي يقوم بها الجسم في وضع الراحة لتعديل الوضعية العامة."; result.descEn = "Non-localized background cortical activity consistent with minor involuntary shifts executed by the body during rest intervals."; } return result; }
